cmake_minimum_required(VERSION 3.16)
project(BackroomsCGF VERSION 0.12.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 REQUIRED)
add_executable(Backrooms backrooms.cpp world.cpp common/threat.cpp common/threat_model.cpp common/audio.cpp common/music.cpp common/Shader.cpp common/ui.cpp common/textures.cpp common/glad.c)
target_include_directories(Backrooms PRIVATE include .)
target_link_libraries(Backrooms PRIVATE glfw OpenGL::GL)
if(WIN32)
  target_link_libraries(Backrooms PRIVATE winmm)
endif()
if(MINGW)
  target_link_options(Backrooms PRIVATE -static-libgcc -static-libstdc++)
endif()
foreach(shader vshader_backrooms.glsl fshader_backrooms.glsl vshader_lens.glsl fshader_lens.glsl)
  add_custom_command(TARGET Backrooms POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${shader}" "$<TARGET_FILE_DIR:Backrooms>/${shader}")
endforeach()
add_custom_command(TARGET Backrooms POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/audio" "$<TARGET_FILE_DIR:Backrooms>/assets/audio")
add_custom_command(TARGET Backrooms POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/models" "$<TARGET_FILE_DIR:Backrooms>/assets/models")
enable_testing()
add_executable(world_tests world.cpp tests/world_tests.cpp)
target_include_directories(world_tests PRIVATE .)
add_test(NAME world_invariants COMMAND world_tests)
add_executable(streaming_tests world.cpp tests/streaming_tests.cpp)
target_include_directories(streaming_tests PRIVATE .)
add_test(NAME streaming_invariants COMMAND streaming_tests)
add_executable(movement_tests tests/movement_tests.cpp)
target_include_directories(movement_tests PRIVATE .)
add_test(NAME movement_invariants COMMAND movement_tests)
add_executable(threat_tests world.cpp common/threat.cpp tests/threat_tests.cpp)
target_include_directories(threat_tests PRIVATE .)
add_test(NAME threat_invariants COMMAND threat_tests)
add_executable(threat_model_tests common/threat_model.cpp tests/threat_model_tests.cpp)
target_include_directories(threat_model_tests PRIVATE .)
add_test(NAME threat_model_invariants COMMAND threat_model_tests)
set_tests_properties(threat_model_invariants PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(encounter_tests tests/encounter_tests.cpp)
target_include_directories(encounter_tests PRIVATE .)
add_test(NAME encounter_invariants COMMAND encounter_tests)
add_executable(audio_tests common/audio.cpp tests/audio_tests.cpp)
target_include_directories(audio_tests PRIVATE .)
if(WIN32)
  target_link_libraries(audio_tests PRIVATE winmm)
endif()
add_test(NAME audio_invariants COMMAND audio_tests)
set_tests_properties(audio_invariants PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
  add_executable(music_tests common/music.cpp tests/music_tests.cpp)
  target_include_directories(music_tests PRIVATE .)
  target_link_libraries(music_tests PRIVATE winmm)
  add_test(NAME music_lifecycle COMMAND music_tests)
endif()
